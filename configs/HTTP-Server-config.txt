! Http-Server - Simulates a remote internet host
! Uses Cisco IOS ip http server to act as a real HTTP target for testing

hostname Http-Server

interface e0/0
 ip address 203.0.113.2 255.255.255.252
 no shutdown

ip route 0.0.0.0 0.0.0.0 203.0.113.1

ip http server
ip http authentication local

end
