! R3 - Office A Edge Router
! Static NAT (PC7) + Dynamic NAT Pool (PC8, PC9, PC10 - only 2 public IPs)

hostname R3

interface e0/0
 description Outside - to ISP
 ip address 203.0.113.18 255.255.255.240
 ip nat outside
 no shutdown

interface e0/1
 description Inside - to SW5
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
 no shutdown

ip route 0.0.0.0 0.0.0.0 203.0.113.17

! ===== Static NAT for PC7 =====
ip nat inside source static 192.168.10.11 203.0.113.19

! ===== Dynamic NAT Pool for PC8, PC9, PC10 =====
ip access-list standard NAT-POOL-ACL
 permit 192.168.10.12 0.0.0.0
 permit 192.168.10.13 0.0.0.0
 permit 192.168.10.14 0.0.0.0

ip nat pool DYNAMIC-POOL 203.0.113.20 203.0.113.21 netmask 255.255.255.240

ip nat inside source list NAT-POOL-ACL pool DYNAMIC-POOL

end
